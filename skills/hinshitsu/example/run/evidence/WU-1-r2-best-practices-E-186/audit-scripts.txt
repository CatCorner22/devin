// The auditor's own arithmetic: BigInt, half-up by comparing the remainder, payments built one person at a time.
function expect(billText, tip, people) {
  const [w, f = ''] = billText.split('.');
  const bill = BigInt(w) * 100n + BigInt((f + '00').slice(0, 2));
  const prod = bill * BigInt(tip);
  let tipC = prod / 100n; if (prod % 100n >= 50n) tipC += 1n;
  const total = bill + tipC;
  const pays = []; let left = total;
  for (let i = 0; i < people; i++) { const share = left / BigInt(people - i); pays.push(share); left -= share; }
  pays.sort((a, b) => (a < b ? 1 : a > b ? -1 : 0));
  return { bill, tipC, total, pays };
}
const fmt = c => { const s = c.toString().padStart(3, '0'); return s.slice(0, -2) + '.' + s.slice(-2); };
function expectedLines(billText, tip, people) {
  const e = expect(billText, tip, people);
  const lines = ['Bill total: ' + fmt(e.bill), 'Tip percent: ' + tip, 'Number of people: ' + people, 'Tip amount: ' + fmt(e.tipC), 'Total with tip: ' + fmt(e.total)];
  const groups = []; for (const p of e.pays) { const g = groups[groups.length - 1]; if (g && g.c === p) g.n++; else groups.push({ c: p, n: 1 }); }
  for (const g of groups) lines.push(g.n === 1 ? '1 person pays ' + fmt(g.c) : g.n + ' people pay ' + fmt(g.c) + ' each');
  if (groups.length === 2) lines.push('Your group decides who pays the higher amount.');
  const sum = e.pays.reduce((a, b) => a + b, 0n);
  if (sum !== e.total || e.pays[0] - e.pays[e.pays.length - 1] > 1n || groups.length > 2) throw new Error('auditor code self-check failed');
  return lines;
}
(async () => {
  const { browser, page } = await launch();
  await page.goto(FILE);
  const cur = {}; let n = 0, bad = 0; const fails = [];
  async function run(b, t, p) {
    const vals = { bill: b, tip: String(t), people: String(p) };
    for (const k of ['bill', 'tip', 'people']) if (cur[k] !== vals[k]) { await page.fill('#' + k, vals[k]); cur[k] = vals[k]; }
    const got = await page.evaluate(() => Array.from(document.querySelectorAll('#result p')).map(x => x.textContent));
    const whole = await page.evaluate(() => document.getElementById('result').textContent);
    const want = expectedLines(b, Number(t), Number(p));
    // page's own payments: sum and spread, read from its text
    let ok = JSON.stringify(got) === JSON.stringify(want) && whole === want.join('');
    n++; if (!ok) { bad++; if (fails.length < 20) fails.push({ b, t, p, got, want }); }
  }
  const cents = c => Math.floor(c / 100) + '.' + String(c % 100).padStart(2, '0');
  let a = 0;
  for (const t of [0, 15, 18, 20]) for (let c = 1; c <= 5000; c++) await run(cents(c), t, 1);
  console.log('sweep A (bills 0.01-50.00 x tips 0,15,18,20 x 1 person): cases', n - a, 'mismatches', bad); a = n; const badA = bad;
  for (let p = 1; p <= 12; p++) for (let c = 1; c <= 500; c++) await run(cents(c), 18, p);
  console.log('sweep B (bills 0.01-5.00 x 1-12 people x tip 18): cases', n - a, 'mismatches', bad - badA); a = n; const badB = bad;
  for (const s of [['10.00', '0', '3'], ['1000000.00', '100', '100'], ['0.01', '0', '100']]) {
    await run(...s);
    console.log('spot', JSON.stringify(s), JSON.stringify(await page.evaluate(() => Array.from(document.querySelectorAll('#result p')).map(x => x.textContent))));
  }
  console.log('spot cases', n - a, 'mismatches', bad - badB);
  console.log('total cases', n, 'total mismatches', bad, JSON.stringify(fails));
  await browser.close();
